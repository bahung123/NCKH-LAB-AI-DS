\chapter{Kết quả áp dụng}\label{ch:ketqua}

\begin{tomtat}[Phạm vi của chương]
Chương này tổng hợp những gì \textbf{đã được xây dựng và đang có trong hệ thống}, căn cứ tài liệu bàn giao.
Các số liệu đo kết quả mô hình và kết quả kinh doanh \textbf{chưa có trong tài liệu nguồn} nên được để trống cho nhóm tác giả điền, không ước tính.
\end{tomtat}

\section{Kết quả về sản phẩm}

\begin{enumerate}
  \item \textbf{Dây chuyền khép kín từ dữ liệu thô đến danh sách khách hàng ưu tiên.} Sáu luồng công việc tự động nối tiếp nhau, từ nạp dữ liệu đến chấm điểm, không cần thao tác tay giữa các bước.
  \item \textbf{Danh sách khách hàng rủi ro cao sẵn sàng theo chu kỳ.} Mỗi lần chạy, hệ thống ghi bảng rủi ro vào cơ sở dữ liệu và xuất tệp CSV để đội chăm sóc dùng ngay; danh sách giới hạn ở nhóm 5\% rủi ro cao nhất, đủ ngắn để xử lý.
  \item \textbf{Mỗi khách hàng có lý do rủi ro.} Tối đa ba lý do, thuộc tám nhóm nghiệp vụ, kèm chỉ số hiện tại, mức nền và mức thay đổi.
  \item \textbf{Cơ chế tự bảo vệ chất lượng.} Mô hình được huấn luyện lại định kỳ (chu kỳ thiết kế khi đủ dữ liệu), chỉ thay thế khi tốt hơn và qua rào chắn an toàn; khi có sự cố, mô hình cũ vẫn tiếp tục phục vụ.
  \item \textbf{Giám sát và truy vết đầy đủ.} Nhật ký từng lần chạy, theo dõi độ lệch dữ liệu và điểm số, dọn dẹp tự động theo quý.
  \item \textbf{Chạy trên hạ tầng sẵn có.} Toàn bộ hệ thống đóng gói theo container, dùng PostgreSQL của đơn vị và có quy trình đóng gói, phát hành, hoàn tác phiên bản.
\end{enumerate}

\begin{luuy}[Về tính trung thực của số liệu]
Việc điền đủ các số liệu trên là điều kiện để đánh giá đúng chất lượng. Nên ghi rõ tháng dữ liệu và phiên bản mô hình mà con số tương ứng, và chỉ dùng kết quả trên bài thi cuối kỳ (không dùng kết quả của các vòng tinh chỉnh) để báo cáo.
\end{luuy}
